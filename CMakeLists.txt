cmake_minimum_required(VERSION 3.20)
project(dsc_keybus_gateway LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

add_executable(digest_guard_tests tests/integration/digest_guard.cpp)
target_compile_features(digest_guard_tests PRIVATE cxx_std_17)
target_include_directories(digest_guard_tests PRIVATE ${CMAKE_SOURCE_DIR})
if(ENABLE_SANITIZERS)
    target_compile_options(digest_guard_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    target_link_options(digest_guard_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME digest_guard COMMAND digest_guard_tests)
add_executable(activity_tests tests/integration/activity.cpp lib/dsc_app/src/activity.cpp)
target_compile_features(activity_tests PRIVATE cxx_std_17)
target_include_directories(activity_tests PRIVATE lib/dsc_app/src lib/dsc_core/src tests/support)
if(ENABLE_SANITIZERS)
    target_compile_options(activity_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    target_link_options(activity_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME observed_activity COMMAND activity_tests)

add_executable(keybus_tests
    tests/keybus_tests.cpp
    tests/support/fake_arduino.cpp
    tests/reference-taligentx/src/dscKeybusInterface.cpp
    tests/reference-taligentx/src/dscKeybusProcessData.cpp
    tests/reference-taligentx/src/dscKeybusPrintData.cpp)
target_compile_features(keybus_tests PRIVATE cxx_std_17)
target_compile_definitions(keybus_tests PRIVATE ESP32)
target_include_directories(keybus_tests PRIVATE tests/support tests/fixtures tests/reference-taligentx/src)
if(ENABLE_SANITIZERS)
    if(MSVC OR WIN32)
        message(FATAL_ERROR "Sanitizer validation is supported on Linux with GCC/Clang.")
    endif()
    target_compile_options(keybus_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    target_link_options(keybus_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME upstream_integrity COMMAND Python3::Interpreter ${CMAKE_SOURCE_DIR}/tools/verify_upstream.py)
add_test(NAME keybus_scenarios COMMAND Python3::Interpreter ${CMAKE_SOURCE_DIR}/tools/run_tests.py --binary $<TARGET_FILE:keybus_tests>)
set_tests_properties(upstream_integrity keybus_scenarios PROPERTIES TIMEOUT 240)

file(GLOB_RECURSE DSC_CORE_SOURCES CONFIGURE_DEPENDS lib/dsc_core/src/*.cpp)
add_executable(owned_core_tests tests/keybus_tests.cpp tests/core/core_regressions.cpp tests/support/fake_arduino.cpp ${DSC_CORE_SOURCES})
target_compile_features(owned_core_tests PRIVATE cxx_std_17)
target_compile_definitions(owned_core_tests PRIVATE ESP32 DSC_OWN_CORE KEYBUS_NO_DIAGNOSTICS)
target_include_directories(owned_core_tests PRIVATE tests/support tests/fixtures lib/dsc_core/src)
if(ENABLE_SANITIZERS)
    target_compile_options(owned_core_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    target_link_options(owned_core_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME owned_core_scenarios COMMAND Python3::Interpreter ${CMAKE_SOURCE_DIR}/tools/run_tests.py --own-core --binary $<TARGET_FILE:owned_core_tests>)
set_tests_properties(owned_core_scenarios PROPERTIES TIMEOUT 240)

file(GLOB DSC_APP_SOURCES CONFIGURE_DEPENDS lib/dsc_app/src/*.cpp)
add_executable(application_tests tests/application/service_tests.cpp tests/support/fake_arduino.cpp
    firmware/simulator/src/scripted_panel.cpp ${DSC_CORE_SOURCES} ${DSC_APP_SOURCES})
target_compile_features(application_tests PRIVATE cxx_std_17)
target_compile_definitions(application_tests PRIVATE ESP32 DSC_ENABLE_REPLAY)
target_include_directories(application_tests PRIVATE tests/support tests/fixtures lib/dsc_core/src lib/dsc_app/src firmware/simulator/src)
target_link_libraries(application_tests PRIVATE Threads::Threads)
if(ENABLE_SANITIZERS)
    target_compile_options(application_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    target_link_options(application_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME application_scenarios COMMAND Python3::Interpreter ${CMAKE_SOURCE_DIR}/tools/run_tests.py --application --binary $<TARGET_FILE:application_tests>)
set_tests_properties(application_scenarios PROPERTIES TIMEOUT 240)
